% Preamble for the random-greedy certified-ratio figures (pgfplots).
% \input{rgplots-preamble} in the preamble; set \rgroot to the path of this latex/ folder
% relative to the main document if it is not the document's directory, e.g.
%   \newcommand{\rgroot}{figures/rg}   before \input{figures/rg/rgplots-preamble}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\providecommand{\rgroot}{.}

% colors: one per upper bound, the same in every plot (colorblind-safe; dashes add redundancy)
\definecolor{rgDualColor}{HTML}{2A78D6}
\definecolor{rgTopkColor}{HTML}{EB6834}
\definecolor{rgTotalColor}{HTML}{1BAF7A}

\newlength{\rgplotwidth}
\setlength{\rgplotwidth}{0.49\columnwidth}

\pgfplotsset{
    rgaxis/.style={
        width=\rgplotwidth, height=0.82\rgplotwidth,
        ymin=0, ymax=1.05, ytick={0,0.2,0.4,0.6,0.8,1},
        xlabel={$k$},
        grid=major, grid style={black!10, line width=0.3pt},
        axis line style={black!60}, tick style={black!60},
        tick label style={font=\scriptsize}, label style={font=\scriptsize},
        title style={font=\scriptsize, yshift=-0.8ex},
        xlabel style={yshift=0.6ex}, ylabel style={yshift=-0.9ex},
        scaled x ticks=false, xticklabel style={/pgf/number format/1000 sep={}},
        unbounded coords=jump, no markers,
        every axis plot/.append style={line width=0.9pt},
        clip mode=individual,
    },
    rgdual/.style={rgDualColor, solid, line width=0.9pt},
    rgtopk/.style={rgTopkColor, dashed, line width=0.9pt},
    rgtotal/.style={rgTotalColor, dash dot, line width=0.9pt},
    rgopt/.style={black, densely dotted, line width=1.1pt},
    rgonee/.style={black!55, dotted, line width=0.7pt},
    rglegendaxis/.style={
        hide axis, xmin=0, xmax=1, ymin=0, ymax=1, scale only axis, width=1pt, height=1pt,
        legend style={draw=none, font=\scriptsize, column sep=0.6em,
                      /tikz/every even column/.append style={column sep=1em}},
        legend cell align=left,
    },
}
